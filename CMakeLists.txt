cmake_minimum_required(VERSION 3.30)
project(PinballDriver LANGUAGES C ASM)
option(PINBALL_ENABLE_HALL_ADC "Sample the Hall sensor inputs" ON)
set(PUYA_SDK_ROOT "C:/Tools/Puya/PY32F002B_Firmware" CACHE PATH "Official PY32F002B firmware library")
set(template "${PUYA_SDK_ROOT}/Templates/PY32F002Bxx_Templates")
set(startup "${template}/EIDE/startup_py32f002bxx.s")
set(linker "${template}/EIDE/py32f002bx5.ld")
foreach(required IN ITEMS "${startup}" "${linker}" "${template}/Src/system_py32f002b.c")
  if(NOT EXISTS "${required}")
    message(FATAL_ERROR "SDK file missing: ${required}. Set PUYA_SDK_ROOT to the extracted official SDK.")
  endif()
endforeach()
add_executable(py32_smoke firmware/py32/src/main.c firmware/py32/src/sk6812.S "${startup}" "${template}/Src/system_py32f002b.c")
set_target_properties(py32_smoke PROPERTIES SUFFIX ".elf" C_STANDARD 11 C_STANDARD_REQUIRED YES)
set_source_files_properties("${startup}" PROPERTIES COMPILE_FLAGS "-x assembler-with-cpp")
target_compile_definitions(py32_smoke PRIVATE PY32F002Bx5 PINBALL_ENABLE_HALL_ADC=$<BOOL:${PINBALL_ENABLE_HALL_ADC}>)
target_include_directories(py32_smoke PRIVATE "${PUYA_SDK_ROOT}/Drivers/CMSIS/Include" "${PUYA_SDK_ROOT}/Drivers/CMSIS/Device/PY32F0xx/Include" "${PUYA_SDK_ROOT}/Drivers/PY32F002B_HAL_Driver/Inc")
target_compile_options(py32_smoke PRIVATE -mcpu=cortex-m0plus -mthumb -Og -g3 -Wall -Wextra -ffunction-sections -fdata-sections)
target_link_options(py32_smoke PRIVATE -mcpu=cortex-m0plus -mthumb --specs=nano.specs --specs=rdimon.specs "-T${linker}" -Wl,--gc-sections "-Wl,-Map=${CMAKE_CURRENT_BINARY_DIR}/py32_smoke.map")
set_property(TARGET py32_smoke APPEND PROPERTY LINK_DEPENDS "${linker}")
add_custom_command(TARGET py32_smoke POST_BUILD
  COMMAND "${CMAKE_OBJCOPY}" -O ihex "$<TARGET_FILE:py32_smoke>" "${CMAKE_CURRENT_BINARY_DIR}/py32_smoke.hex"
  COMMAND "${CMAKE_OBJCOPY}" -O binary "$<TARGET_FILE:py32_smoke>" "${CMAKE_CURRENT_BINARY_DIR}/py32_smoke.bin"
  COMMAND "${CMAKE_SIZE}" "$<TARGET_FILE:py32_smoke>" VERBATIM)
